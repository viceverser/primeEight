 if(NOT EXISTS "${PRIMEEIGHT_CLAP_INCLUDE_DIR}/clap/clap.h")
  message(FATAL_ERROR "Official CLAP headers missing. Set PRIMEEIGHT_CLAP_INCLUDE_DIR to the official clap/include directory.")
 endif()
 if(PRIMEEIGHT_BUILD_CLAP_JUCE_SHIM)
  if(NOT WIN32)
   message(FATAL_ERROR "PrimeEight experimental JUCE shim is Windows-only at this stage")
  endif()
  if(NOT EXISTS "${PRIMEEIGHT_SST_CLAP_HELPERS_DIR}/CMakeLists.txt")
   message(FATAL_ERROR "Set PRIMEEIGHT_SST_CLAP_HELPERS_DIR to sst-clap-helpers")
  endif()
  if(NOT EXISTS "${PRIMEEIGHT_JUCE_DIR}/CMakeLists.txt")
   message(FATAL_ERROR "Set PRIMEEIGHT_JUCE_DIR to JUCE")
  endif()
  # The shim's CMake helper requires the official CLAP target.
  add_library(clap-core INTERFACE)
  target_include_directories(clap-core INTERFACE "${PRIMEEIGHT_CLAP_INCLUDE_DIR}")
  add_subdirectory("${PRIMEEIGHT_SST_CLAP_HELPERS_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/sst-clap-helpers")
  add_clap_juce_shim(JUCE_PATH "${PRIMEEIGHT_JUCE_DIR}")
 endif()
 add_library(PrimeEight_CLAP MODULE clap/PrimeEightClap.cpp)
 target_link_libraries(PrimeEight_CLAP PRIVATE primeeight_dsp)
 if(PRIMEEIGHT_BUILD_CLAP_JUCE_SHIM)
  target_sources(PrimeEight_CLAP PRIVATE gui/PrimeEightClapJuce.cpp)
  target_compile_features(PrimeEight_CLAP PRIVATE cxx_std_20)
  target_compile_definitions(PrimeEight_CLAP PRIVATE PRIMEEIGHT_USE_JUCE_SHIM=1)
  target_include_directories(PrimeEight_CLAP PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/gui")
  target_link_libraries(PrimeEight_CLAP PRIVATE sst::clap_juce_shim sst::clap_juce_shim_headers juce::juce_gui_basics)
 endif()
 target_include_directories(PrimeEight_CLAP PRIVATE "${PRIMEEIGHT_CLAP_INCLUDE_DIR}")
 if(WIN32)
  target_link_libraries(PrimeEight_CLAP PRIVATE user32 gdi32)
 endif()
 set_target_properties(PrimeEight_CLAP PROPERTIES PREFIX "" SUFFIX ".clap")
endif()

# to validate the GUI toolchain independently before embedding it into CLAP.
option(PRIMEEIGHT_BUILD_JUCE_PREVIEW "Build standalone JUCE vector editor preview" OFF)
set(PRIMEEIGHT_JUCE_DIR "" CACHE PATH "JUCE source directory containing CMakeLists.txt")
if(PRIMEEIGHT_BUILD_JUCE_PREVIEW)

  message(FATAL_ERROR "JUCE source missing: set PRIMEEIGHT_JUCE_DIR to a JUCE checkout")
 endif()
 add_subdirectory("${PRIMEEIGHT_JUCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/juce")
 juce_add_gui_app(PrimeEight_JUCE_Preview VERSION 0.1.33 PRODUCT_NAME "PrimeEight JUCE Preview")
 target_sources(PrimeEight_JUCE_Preview PRIVATE gui/JuceEditorPreview.cpp)
 target_compile_features(PrimeEight_JUCE_Preview PRIVATE cxx_std_17)

  JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)

  juce::juce_gui_extra juce::juce_recommended_config_flags
  juce::juce_recommended_warning_flags)
