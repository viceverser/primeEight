cmake_minimum_required(VERSION 3.16)
project(PrimeEight LANGUAGES CXX)
add_library(primeeight_dsp INTERFACE)
target_include_directories(primeeight_dsp INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_compile_features(primeeight_dsp INTERFACE cxx_std_17)
add_executable(primeeight_regression tests/regression.cpp)
target_link_libraries(primeeight_regression PRIVATE primeeight_dsp)
add_executable(primeeight_processor_test tests/processor.cpp)
target_link_libraries(primeeight_processor_test PRIVATE primeeight_dsp)
enable_testing()
add_test(NAME core_regression COMMAND primeeight_regression)
add_test(NAME processor_bridge COMMAND primeeight_processor_test)
# Actual VST3 target requires Steinberg VST3 SDK and is not falsely represented as built.
add_executable(primeeight_contract_test tests/contract.cpp)
target_link_libraries(primeeight_contract_test PRIVATE primeeight_dsp)
add_test(NAME plugin_contract COMMAND primeeight_contract_test)
# Both actual plugin wrappers require their respective official SDKs.
# Never substitute an ABI guess for a loadable plugin.
option(PRIMEEIGHT_BUILD_CLAP "Build CLAP adapter when official headers are installed" OFF)
option(PRIMEEIGHT_BUILD_VST3 "Build VST3 adapter when Steinberg SDK is installed" OFF)
set(PRIMEEIGHT_CLAP_INCLUDE_DIR "" CACHE PATH "Official CLAP include directory (contains clap/clap.h)")
if(PRIMEEIGHT_BUILD_CLAP)
 if(NOT EXISTS "${PRIMEEIGHT_CLAP_INCLUDE_DIR}/clap/clap.h")
  message(FATAL_ERROR "Official CLAP headers missing. Set PRIMEEIGHT_CLAP_INCLUDE_DIR to the official clap/include directory.")
 endif()
 add_library(PrimeEight_CLAP MODULE clap/PrimeEightClap.cpp)
 target_link_libraries(PrimeEight_CLAP PRIVATE primeeight_dsp)
 target_include_directories(PrimeEight_CLAP PRIVATE "${PRIMEEIGHT_CLAP_INCLUDE_DIR}")
 set_target_properties(PrimeEight_CLAP PROPERTIES PREFIX "" SUFFIX ".clap")
endif()
if(PRIMEEIGHT_BUILD_VST3)
 message(FATAL_ERROR "VST3 SDK adapter is not yet implemented.")
endif()
